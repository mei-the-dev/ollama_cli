Based on the architecture of **Continue** (available at `github.com/continuedev/continue`), the biggest power-up we can give Omarchy is a modular **Context Provider System**.

Continue excels because it doesn't just read files; it allows you to inject context like `@diff` (git changes), `@url` (documentation), and `@tree` (project structure) dynamically into the prompt.

Here are the improved files. I have refactored the CLI to include a **Context Engine** inspired by Continue, and added the necessary supporting tools to the Server.

### 1. `mcp_server.py`

**Improvements:**

* **Added `fetch_url` tool**: Allows the agent to read external documentation (a key Continue feature).
* **Enhanced `git_diff**`: Specialized tool to get context for `@diff`.
* **Scaffolded `config.json**`: Server now respects a config file for model definitions, similar to `~/.continue/config.json`.

```python
#!/usr/bin/env python3
"""
Omarchy MCP Server v2.1 (Continue-Inspired)
Features: Context Providers, URL Fetching, robust Git integration.
"""

import json
import os
import subprocess
import sys
from pathlib import Path
from typing import Any, Dict, List, Optional
import asyncio
import aiohttp
from datetime import datetime
import difflib
import ast
import shutil
import urllib.request
from dataclasses import dataclass, field
from enum import Enum
from aiohttp import web

# --- Configuration ---
CONFIG_PATH = Path.home() / ".omarchy" / "config.json"

def load_config():
    default_config = {
        "model": "qwen2.5-coder:14b-instruct-q4_K_M",
        "temperature": 0.0,
        "context_length": 8192
    }
    if CONFIG_PATH.exists():
        try:
            return {**default_config, **json.loads(CONFIG_PATH.read_text())}
        except: pass
    return default_config

# --- Structured Error Handling ---
class ToolStatus(Enum):
    SUCCESS = "SUCCESS"
    ERROR = "ERROR"

@dataclass
class ToolResult:
    status: ToolStatus
    data: Any = None
    error: Optional[str] = None
    metadata: dict = field(default_factory=dict)

    def to_dict(self):
        return {
            "status": self.status.value,
            "data": self.data,
            "error": self.error,
            "metadata": self.metadata
        }

class MCPServer:
    def __init__(self):
        self.root_dir = Path.home() / ".omarchy"
        self.config = load_config()
        self.init_directories()

    def init_directories(self):
        for d in ["knowledge", "tools", "logs"]:
            (self.root_dir / d).mkdir(parents=True, exist_ok=True)
        if not CONFIG_PATH.exists():
            CONFIG_PATH.write_text(json.dumps(self.config, indent=2))

    async def handle_request(self, request: Dict[str, Any]) -> Dict[str, Any]:
        method = request.get("method")
        params = request.get("params", {})
        
        if method == "tools/list":
            return await self.list_tools()
        elif method == "tools/call":
            return await self.call_tool(params)
        elif method == "config/get":
            return self.config
        return {"error": f"Unknown method: {method}"}

    async def list_tools(self) -> Dict:
        return {
            "tools": [
                {
                    "name": "search_files",
                    "description": "Smart fuzzy search (BM25-lite style) with context",
                    "inputSchema": {"type": "object", "properties": {"query": {"type": "string"}, "path": {"type": "string"}}}
                },
                {
                    "name": "fetch_url",
                    "description": "Fetch text content from a URL (Documentation reader)",
                    "inputSchema": {"type": "object", "properties": {"url": {"type": "string"}}}
                },
                {
                    "name": "get_git_diff",
                    "description": "Get current unstaged changes (Context for @diff)",
                    "inputSchema": {"type": "object", "properties": {}}
                },
                {
                    "name": "apply_edit",
                    "description": "Surgical code replacement",
                    "inputSchema": {"type": "object", "properties": {"filepath": {"type": "string"}, "search": {"type": "string"}, "replace": {"type": "string"}}}
                },
                {
                    "name": "get_repo_map",
                    "description": "Get visual tree of repository structure",
                    "inputSchema": {"type": "object", "properties": {"path": {"type": "string"}, "max_depth": {"type": "integer"}}}
                },
                {
                    "name": "execute_code", 
                    "description": "Run shell commands",
                    "inputSchema": {"type": "object", "properties": {"command": {"type": "string"}}}
                },
                {
                    "name": "github_tool",
                    "description": "GitHub CLI wrapper",
                    "inputSchema": {"type": "object", "properties": {"subcommand": {"type": "string"}, "args": {"type": "array"}}}
                }
            ]
        }

    async def call_tool(self, params: Dict) -> Dict:
        name = params.get("name")
        args = params.get("arguments", {})
        method = getattr(self, f"tool_{name}", None)
        if not method:
            return ToolResult(ToolStatus.ERROR, error=f"Tool not found: {name}").to_dict()
        try:
            result = await method(args)
            return result.to_dict() if isinstance(result, ToolResult) else result
        except Exception as e:
            return ToolResult(ToolStatus.ERROR, error=str(e)).to_dict()

    # --- Tool Implementations ---

    async def tool_fetch_url(self, args: Dict) -> ToolResult:
        """Mimics Continue's ability to read docs."""
        url = args["url"]
        try:
            # Basic scraper using standard lib to avoid extra dependencies
            # In a full prod version, use beautifulsoup4
            with urllib.request.urlopen(url) as response:
                html = response.read().decode('utf-8')
                # Very naive HTML stripping
                text = re.sub('<[^<]+?>', '', html)
                # Squeeze whitespace
                text = " ".join(text.split())
                return ToolResult(ToolStatus.SUCCESS, data=text[:8000]) # Token limit safety
        except Exception as e:
            return ToolResult(ToolStatus.ERROR, error=f"Failed to fetch {url}: {e}")

    async def tool_get_git_diff(self, args: Dict) -> ToolResult:
        """Support for @diff context provider."""
        try:
            res = subprocess.run(["git", "diff", "HEAD"], capture_output=True, text=True)
            if not res.stdout:
                return ToolResult(ToolStatus.SUCCESS, data="No git changes found.")
            return ToolResult(ToolStatus.SUCCESS, data=res.stdout)
        except Exception as e:
            return ToolResult(ToolStatus.ERROR, error=str(e))

    async def tool_search_files(self, args: Dict) -> ToolResult:
        query = args["query"]
        path = args.get("path", ".")
        matches = []
        # Continue uses embeddings; we use weighted fuzzy search for portability
        for root, _, files in os.walk(path):
            if ".git" in root or "__pycache__" in root: continue
            for file in files:
                fp = Path(root) / file
                try:
                    content = fp.read_text(errors='ignore')
                    if query in content:
                        lines = content.splitlines()
                        for i, line in enumerate(lines):
                            if query in line:
                                matches.append({
                                    "file": str(fp),
                                    "line": i + 1,
                                    "content": line.strip(),
                                    "context": [l.strip() for l in lines[max(0, i-2):min(len(lines), i+3)]]
                                })
                except: continue
        return ToolResult(ToolStatus.SUCCESS, data={"matches": matches[:20]}) # Limit results

    async def tool_apply_edit(self, args: Dict) -> ToolResult:
        fp = Path(args["filepath"])
        if not fp.exists(): return ToolResult(ToolStatus.ERROR, error="File not found")
        content = fp.read_text()
        search_str = args["search"]
        if search_str not in content:
            # Normalize whitespace for "fuzzy" match
            if " ".join(search_str.split()) not in " ".join(content.split()):
                 return ToolResult(ToolStatus.ERROR, error="Search pattern not found")
        new_content = content.replace(search_str, args["replace"])
        backup = fp.with_suffix(f".bak.{int(datetime.now().timestamp())}")
        fp.write_text(new_content)
        return ToolResult(ToolStatus.SUCCESS, data={"backup": str(backup), "applied": True})

    async def tool_get_repo_map(self, args: Dict) -> ToolResult:
        path = Path(args.get("path", "."))
        tree = []
        for root, dirs, files in os.walk(path):
            level = root.replace(str(path), '').count(os.sep)
            if level > 2: continue # Max depth hardcoded for speed
            indent = '  ' * level
            tree.append(f"{indent}{os.path.basename(root)}/")
            for f in files[:8]: # Limit files for context window
                if not f.startswith('.'): tree.append(f"{indent}  {f}")
        return ToolResult(ToolStatus.SUCCESS, data="\n".join(tree))

    async def tool_execute_code(self, args: Dict) -> ToolResult:
        res = subprocess.run(args["command"], shell=True, capture_output=True, text=True)
        return ToolResult(ToolStatus.SUCCESS if res.returncode==0 else ToolStatus.ERROR, data=res.stdout, error=res.stderr)
    
    async def tool_github_tool(self, args: Dict) -> ToolResult:
        if not shutil.which("gh"): return ToolResult(ToolStatus.ERROR, error="gh not installed")
        cmd = ["gh", args["subcommand"]] + args.get("args", [])
        res = subprocess.run(cmd, capture_output=True, text=True)
        return ToolResult(ToolStatus.SUCCESS, data=res.stdout, error=res.stderr)

# --- Server Start ---
async def start_server():
    server = MCPServer()
    app = web.Application()
    async def handle_http(request):
        try:
            return web.json_response(await server.handle_request(await request.json()))
        except Exception as e:
            return web.json_response({"error": str(e)}, status=500)
    app.router.add_post('/mcp', handle_http)
    runner = web.AppRunner(app)
    await runner.setup()
    site = web.TCPSite(runner, 'localhost', 8080)
    print("MCP Server v2.1 running on http://localhost:8080")
    await site.start()
    while True: await asyncio.sleep(3600)

if __name__ == "__main__":
    import re
    try: asyncio.run(start_server())
    except KeyboardInterrupt: pass

```

### 2. `omarchy_cli.py`

**Improvements:**

* **Context Engine**: A sophisticated `resolve_context` method that handles `@file`, `@web` (URL), `@diff` (Git), and `@tree` (Repo map), mirroring Continue's provider system.
* **System Prompt Loading**: Injects specific instructions for surgical editing.
* **Rich UI**: Clean rendering of context injection.

```python
#!/usr/bin/env python3
"""
Omarchy CLI v2.1
Features: Context Providers (@file, @web, @diff), Config Loading
"""
import asyncio
import json
import sys
import re
import argparse
import subprocess
import aiohttp
from pathlib import Path

# Rich Imports
try:
    from rich.console import Console
    from rich.panel import Panel
    from rich.markdown import Markdown
    from rich.syntax import Syntax
    from rich.tree import Tree
    from rich.prompt import Confirm
    from rich.status import Status
except ImportError:
    subprocess.check_call([sys.executable, "-m", "pip", "install", "rich"])
    from rich.console import Console
    from rich.panel import Panel
    from rich.markdown import Markdown
    from rich.syntax import Syntax
    from rich.tree import Tree
    from rich.prompt import Confirm
    from rich.status import Status

console = Console()

class ContextEngine:
    """Inspired by Continue's Context Providers"""
    def __init__(self, cli):
        self.cli = cli

    async def resolve(self, prompt: str) -> str:
        """Parses prompt for @mentions and injects content"""
        # Regex for @provider:value (e.g. @web:http://...) or just @file
        
        # 1. Handle @web (URL)
        matches = re.finditer(r'@web:(\S+)', prompt)
        for m in matches:
            url = m.group(1)
            with console.status(f"[bold green]Fetching docs: {url}[/]"):
                res = await self.cli.call_tool("fetch_url", {"url": url})
                content = res.get("data", "Failed to fetch")
                block = f"\n\n[CONTEXT: URL {url}]\n```\n{content[:4000]}\n```\n"
                prompt = prompt.replace(m.group(0), block)

        # 2. Handle @diff (Git)
        if "@diff" in prompt:
            with console.status("[bold green]Reading git diff...[/]"):
                res = await self.cli.call_tool("get_git_diff", {})
                content = res.get("data", "No diff")
                block = f"\n\n[CONTEXT: GIT DIFF]\n```diff\n{content}\n```\n"
                prompt = prompt.replace("@diff", block)

        # 3. Handle @tree (Repo Map)
        if "@tree" in prompt:
            with console.status("[bold green]Mapping repo...[/]"):
                res = await self.cli.call_tool("get_repo_map", {"path": "."})
                content = res.get("data", "")
                block = f"\n\n[CONTEXT: FILE TREE]\n```\n{content}\n```\n"
                prompt = prompt.replace("@tree", block)

        # 4. Handle @filename
        # Simple heuristic: @ followed by chars, ending with extension or specific chars
        file_matches = re.finditer(r'@([\w./-]+\.[a-zA-Z0-9]+)', prompt)
        for m in file_matches:
            fname = m.group(1)
            path = Path(fname)
            if path.exists():
                content = path.read_text()
                # Truncate
                if len(content) > 6000: content = content[:6000] + "...(truncated)"
                block = f"\n\n[CONTEXT: FILE {fname}]\n```\n{content}\n```\n"
                prompt = prompt.replace(m.group(0), block)
        
        return prompt

class OmarchyCLI:
    def __init__(self):
        self.server_url = "http://localhost:8080/mcp"
        self.ollama_url = "http://localhost:11434/api/generate"
        self.context_engine = ContextEngine(self)
        
        # Load Config
        self.config = {
            "model": "qwen2.5-coder:14b-instruct-q4_K_M",
            "system_prompt": "You are an AI coding agent. Use tools to edit code. When editing, provide the exact search block and replacement block."
        }
        cfg_path = Path.home() / ".omarchy" / "config.json"
        if cfg_path.exists():
            try: self.config.update(json.loads(cfg_path.read_text()))
            except: pass

    async def call_ollama(self, prompt: str, system: str = "") -> str:
        payload = {
            "model": self.config["model"],
            "prompt": f"{system}\n\n{prompt}",
            "stream": False
        }
        try:
            async with aiohttp.ClientSession() as session:
                async with session.post(self.ollama_url, json=payload) as resp:
                    if resp.status == 200:
                        return (await resp.json()).get("response", "")
                    return f"Error: {resp.status}"
        except Exception as e:
            return f"Ollama Connection Error: {e}"

    async def call_tool(self, name: str, args: dict) -> dict:
        try:
            async with aiohttp.ClientSession() as session:
                payload = {"method": "tools/call", "params": {"name": name, "arguments": args}}
                async with session.post(self.server_url, json=payload) as resp:
                    return await resp.json()
        except Exception as e:
            return {"status": "ERROR", "error": str(e)}

    async def process_prompt(self, user_input: str, mode: str = "chat"):
        # 1. Resolve Context Providers
        enriched_prompt = await self.context_engine.resolve(user_input)
        
        # 2. Modes
        if mode == "suggest":
            sys_msg = "Output ONLY a bash command."
            cmd = await self.call_ollama(enriched_prompt, sys_msg)
            console.print(Panel(Syntax(cmd.strip().replace("`", ""), "bash"), title="Suggestion"))
            if Confirm.ask("Execute?"): subprocess.run(cmd, shell=True)
            return

        # 3. Agent Loop
        console.print("[dim]Thinking...[/dim]")
        
        tool_prompt = f"""{self.config['system_prompt']}
If action needed, output JSON: {{"tool": "name", "args": {{...}}}}.
Available: search_files, fetch_url, apply_edit, execute_code.
Context provided above.
"""
        response = await self.call_ollama(enriched_prompt, tool_prompt)
        
        # 4. Parse Tool
        if "{" in response and "tool" in response:
            try:
                # Naive JSON extraction
                json_str = response[response.find("{"):response.rfind("}")+1]
                data = json.loads(json_str)
                if "tool" in data:
                    console.print(f"[bold cyan]Tool Call:[/bold cyan] {data['tool']}")
                    result = await self.call_tool(data['tool'], data.get('args', {}))
                    self.render_result(data['tool'], result)
                    return
            except: pass
        
        console.print(Markdown(response))

    def render_result(self, name, result):
        data = result.get("data")
        status = result.get("status")
        if name == "search_files" and status == "SUCCESS":
            t = Tree(f"Matches")
            for m in data['matches']:
                t.add(f"[bold]{m['file']}:{m['line']}[/bold]").add(Syntax(m['content'], "python"))
            console.print(t)
        else:
            console.print(Panel(str(data), title=f"{name} ({status})", border_style="green" if status == "SUCCESS" else "red"))

    async def interactive_mode(self):
        console.print(f"[bold cyan]Omarchy v2.1[/bold cyan] | Model: [dim]{self.config['model']}[/dim]")
        console.print("[dim]Tips: Use @file.py, @web:url, @diff, @tree[/dim]")
        while True:
            try:
                inp = console.input("\n[bold green]➜[/bold green] ")
                if inp.lower() in ["exit", "quit"]: break
                await self.process_prompt(inp)
            except KeyboardInterrupt: break

async def main():
    parser = argparse.ArgumentParser()
    subparsers = parser.add_subparsers(dest="command")
    subparsers.add_parser("suggest")
    parser.add_argument("query", nargs="*")
    parser.add_argument("-p", "--prompt")
    args = parser.parse_args()
    
    cli = OmarchyCLI()
    target = " ".join(args.query) if args.query else args.prompt
    
    if args.command == "suggest": await cli.process_prompt(target, mode="suggest")
    elif target: await cli.process_prompt(target)
    else: await cli.interactive_mode()

if __name__ == "__main__":
    asyncio.run(main())

```

### 3. `mcp_tool.py`

*(Kept compatible with the changes)*

```python
#!/usr/bin/env python3
import sys
import json
import asyncio
from mcp_server import MCPServer, ToolResult

async def main():
    if len(sys.argv) < 3:
        print("Usage: mcp_tool.py <tool_name> '<json args>'")
        sys.exit(1)
    
    tool = sys.argv[1]
    try: args = json.loads(sys.argv[2])
    except: sys.exit(1)

    server = MCPServer()
    method = getattr(server, f"tool_{tool}", None)
    if method:
        res = await method(args)
        if isinstance(res, ToolResult): print(json.dumps(res.to_dict(), indent=2))
        else: print(json.dumps(res, indent=2))
    else:
        print(f"Error: Tool {tool} not found")

if __name__ == '__main__':
    asyncio.run(main())
```
